cmake_minimum_required(VERSION 3.22)
project(tirsolador LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream melon-build EXCLUDE_FROM_ALL)
if(ANDROID)
  add_library(tirso_retro SHARED retro_host.cpp)
  target_link_libraries(tirso_retro PRIVATE log android EGL GLESv2 dl)
  target_link_options(tirso_retro PRIVATE "-Wl,-z,max-page-size=16384")
  add_library(tirso_ds SHARED engine.cpp jni.cpp)
  target_link_libraries(tirso_ds PRIVATE core log android dl)
  target_link_options(tirso_ds PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(core_test engine.cpp core_test.cpp)
  target_link_libraries(core_test PRIVATE core pthread dl)
endif()
